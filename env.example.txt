# ============================================
# AUTHENTICATION CONFIGURATION
# ============================================
# This file contains all environment variables related to authentication
# Copy relevant settings to your .env file

# --------------------------------------------
# General Settings
# --------------------------------------------
NODE_ENV=development  # development | production | staging

# --------------------------------------------
# JWT Configuration
# --------------------------------------------
# JWT secrets - MUST be different and strong in production!
JWT_SECRET=your-super-secret-jwt-key-change-this-in-production
JWT_ACCESS_SECRET=your-access-token-secret-change-this
JWT_REFRESH_SECRET=your-refresh-token-secret-change-this

# Token expiration times
JWT_ACCESS_TOKEN_EXPIRES_IN=15m   # 15 minutes
JWT_REFRESH_TOKEN_EXPIRES_IN=7d   # 7 days

# --------------------------------------------
# Password Hashing
# --------------------------------------------
BCRYPT_SALT_ROUNDS=12             # Higher = more secure but slower
PASSWORD_MIN_LENGTH=8             # Minimum password length
PASSWORD_BLOCK_COMMON=true        # Block common/weak passwords

# --------------------------------------------
# OTP Configuration
# --------------------------------------------
# OTP generation and validation
OTP_LENGTH=6                      # Number of digits in OTP
OTP_TTL_SECONDS=300               # 5 minutes - OTP validity
OTP_MAX_ATTEMPTS=3                # Max attempts before OTP is invalidated

# OTP rate limiting
OTP_RESEND_COOLDOWN_SECONDS=60    # 1 minute - Wait time between resends
OTP_DAILY_PHONE_LIMIT=5           # Max OTPs per phone per day
OTP_DAILY_IP_LIMIT=20             # Max OTPs per IP per day

# --------------------------------------------
# SMS Configuration
# --------------------------------------------
# SMS provider mode
SMS_PROVIDER=console              # console = log to console (dev)
                                  # api = send via external API (prod)
                                  # If not set, defaults to 'console' in development, 'api' in production

# SMS API settings (https://api.sms.net.bd)
SMS_API_URL=https://api.sms.net.bd/sendsms
SMS_API_KEY=your-sms-api-key-here

# SMS behavior
SMS_THROW_ON_FAILURE=false        # true = throw error if SMS fails (production)
                                  # false = fail silently (development)

# --------------------------------------------
# Account Security
# --------------------------------------------
# Failed login attempts and account lockout
AUTH_ACCOUNT_LOCKOUT_THRESHOLD=5          # Failed attempts before lockout
AUTH_ACCOUNT_LOCKOUT_DURATION_MINUTES=30  # Lockout duration in minutes

# Token expiration
AUTH_REFRESH_TOKEN_EXPIRATION_DAYS=7      # Refresh token validity (days)
AUTH_PASSWORD_RESET_TOKEN_EXPIRATION_HOURS=1  # Password reset token validity

# --------------------------------------------
# Redis Configuration (for caching & rate limiting)
# --------------------------------------------
REDIS_HOST=localhost
REDIS_PORT=6379
REDIS_PASSWORD=changeme           # MUST change in production!
REDIS_DB=0                        # Database number
REDIS_KEY_PREFIX_AUTH=auth:       # Prefix for auth-related keys
REDIS_URL=redis://:changeme@localhost:6379/0  # Full connection URL (alternative)

# --------------------------------------------
# Rate Limiting (API-level)
# --------------------------------------------
# Global API rate limiting
RATE_LIMIT_ENABLED=true           # Enable/disable rate limiting
RATE_LIMIT_TTL=60                 # Window in seconds
RATE_LIMIT_MAX=100                # Max requests per window

# IP-based rate limiting (for anonymous users)
RATE_LIMIT_IP_LIMIT=50            # Requests per window
RATE_LIMIT_IP_WINDOW_SECONDS=60   # Window duration

# User-based rate limiting (for authenticated users)
RATE_LIMIT_USER_LIMIT=100         # Requests per window
RATE_LIMIT_USER_WINDOW_SECONDS=60 # Window duration

# --------------------------------------------
# Database Configuration
# --------------------------------------------
# PostgreSQL connection
DATABASE_HOST=localhost
DATABASE_PORT=5432
DATABASE_NAME=scol_db
DATABASE_USER=postgres
DATABASE_PASSWORD=your-db-password
DATABASE_SSL=false                # true in production

# --------------------------------------------
# Logging
# --------------------------------------------
LOG_LEVEL=debug                   # debug | info | warn | error
LOG_PRETTY=true                   # true in dev, false in prod (JSON logs)

# ============================================
# PRODUCTION-SPECIFIC SETTINGS
# ============================================
# When deploying to production, make sure to:
# 
# 1. Change all secrets/keys to strong random values
# 2. Set SMS_THROW_ON_FAILURE=true
# 3. Set NODE_ENV=production
# 4. Set DATABASE_SSL=true
# 5. Set LOG_PRETTY=false
# 6. Use strong BCRYPT_SALT_ROUNDS (12+)
# 7. Configure proper Redis with password
# 8. Review and adjust rate limits based on traffic
# 9. Enable monitoring and alerting
# 10. Set up proper backup for database

# ============================================
# SECURITY BEST PRACTICES
# ============================================
# 
# ✅ Never commit .env file to version control
# ✅ Use different secrets for dev/staging/production
# ✅ Rotate secrets periodically
# ✅ Use strong random values for JWT secrets (at least 32 characters)
# ✅ Keep SMS API keys secure
# ✅ Monitor failed login attempts
# ✅ Review rate limiting logs regularly
# ✅ Set up alerts for suspicious activity
# ✅ Use HTTPS in production
# ✅ Enable database SSL in production
# ✅ Regular security audits
